// @ts-ignore
import React, { useState, useEffect } from 'react';
import { useNavigate, Link } from 'react-router-dom';
import { RootBlock } from '../models/block';
import { BlockService } from '../services/blockService';

const BlocksPage: React.FC = () => {
    const navigate = useNavigate();
    const [blocks, setBlocks] = useState<RootBlock[]>([]);
    const [loading, setLoading] = useState<boolean>(true);
    const [error, setError] = useState<string | null>(null);
    const [cursor, setCursor] = useState<string | null>(null);
    const [hasMore, setHasMore] = useState<boolean>(true);
    const [loadingMore, setLoadingMore] = useState<boolean>(false);

    const handleNavigateBack = () => {
        navigate('/');
    };

    // @ts-ignore
    const fetchBlocks = async (nextCursor?: string, append: boolean = false) => {
        try {
            if (!append) {
                setLoading(true);
            } else {
                setLoadingMore(true);
            }

            setError(null);

            const fetchedBlocks = await BlockService.getRootBlocks({
                cursor: nextCursor || undefined,
                limit: 10
            });

            if (append) {
                setBlocks(prev => [...prev, ...fetchedBlocks]);
            } else {
                setBlocks(fetchedBlocks);
            }

            // Set cursor for next page (use the last block's cursor)
            if (fetchedBlocks.length > 0) {
                setCursor(fetchedBlocks[fetchedBlocks.length - 1].cursor);
                setHasMore(fetchedBlocks.length === 10); // Assume no more if less than page size
            } else {
                setHasMore(false);
            }

        } catch (err) {
            setError(err instanceof Error ? err.message : 'An error occurred');
        } finally {
            setLoading(false);
            setLoadingMore(false);
        }
    };

    useEffect(() => {
        fetchBlocks();
    }, []);

    const handleLoadMore = () => {
        if (cursor && !loadingMore) {
            fetchBlocks(cursor, true);
        }
    };

    const handleRefresh = () => {
        setCursor(null);
        setHasMore(true);
        fetchBlocks();
    };

    if (loading && blocks.length === 0) {
        return (
            <div style={styles.container}>
                <div style={styles.header}>
                    <button onClick={handleNavigateBack} style={styles.backButton}>
                        ← Back to Home
                    </button>
                    <h1 style={styles.title}>Visual Flow Blocks</h1>
                </div>
                <div style={styles.loadingContainer}>
                    <div style={styles.spinner}></div>
                    <p style={styles.loadingText}>Loading blocks...</p>
                </div>
            </div>
        );
    }

    if (error && blocks.length === 0) {
        return (
            <div style={styles.container}>
                <div style={styles.header}>
                    <button onClick={handleNavigateBack} style={styles.backButton}>
                        ← Back to Home
                    </button>
                    <h1 style={styles.title}>Visual Flow Blocks</h1>
                </div>
                <div style={styles.errorContainer}>
                    <div style={styles.errorIcon}>⚠️</div>
                    <h3 style={styles.errorTitle}>Failed to Load Blocks</h3>
                    <p style={styles.errorMessage}>{error}</p>
                    <button onClick={handleRefresh} style={styles.retryButton}>
                        Try Again
                    </button>
                </div>
            </div>
        );
    }

    return (
        <div style={styles.container}>
            <div style={styles.header}>
                <button onClick={handleNavigateBack} style={styles.backButton}>
                    ← Back to Home
                </button>
                <div style={styles.headerContent}>
                    <h1 style={styles.title}>Visual Flow Blocks</h1>
                    <button onClick={handleRefresh} style={styles.refreshButton} disabled={loading}>
                        🔄 Refresh
                    </button>
                </div>
            </div>

            {blocks.length === 0 ? (
                <div style={styles.emptyContainer}>
                    <div style={styles.emptyIcon}>📭</div>
                    <h3 style={styles.emptyTitle}>No Blocks Found</h3>
                    <p style={styles.emptyMessage}>
                        No flow blocks have been created yet. Start logging with VFL to see them here.
                    </p>
                </div>
            ) : (
                <>
                    <div style={styles.blocksGrid}>
                        {blocks.map((block) => (
                            <div key={block.id} style={styles.blockCard}>
                                <div style={styles.blockHeader}>
                                    <h3 style={styles.blockName}>{block.name}</h3>
                                    <span style={styles.blockId}>#{block.id.slice(0, 8)}</span>
                                </div>

                                <div style={styles.blockMetrics}>
                                    <div style={styles.metric}>
                                        <span style={styles.metricLabel}>Duration:</span>
                                        <span style={styles.metricValue}>
                      {BlockService.formatDuration(block.startTime, block.endTime)}
                    </span>
                                    </div>

                                    <div style={styles.metric}>
                                        <span style={styles.metricLabel}>Started:</span>
                                        <span style={styles.metricValue}>
                      {BlockService.formatDateTime(block.startTime)}
                    </span>
                                    </div>

                                    <div style={styles.metric}>
                                        <span style={styles.metricLabel}>Created:</span>
                                        <span style={styles.metricValue}>
                      {BlockService.formatDateTime(block.createdAt)}
                    </span>
                                    </div>
                                </div>

                                {block.endMessage && (
                                    <div style={styles.endMessage}>
                                        <span style={styles.endMessageLabel}>End Message:</span>
                                        <p style={styles.endMessageText}>{block.endMessage}</p>
                                    </div>
                                )}

                                <div style={styles.blockActions}>
                                    <button style={styles.viewButton}>
                                        View Details →
                                    </button>
                                </div>
                            </div>
                        ))}
                    </div>

                    {hasMore && (
                        <div style={styles.loadMoreContainer}>
                            <button
                                onClick={handleLoadMore}
                                style={styles.loadMoreButton}
                                disabled={loadingMore}
                            >
                                {loadingMore ? 'Loading...' : 'Load More Blocks'}
                            </button>
                        </div>
                    )}

                    {loadingMore && (
                        <div style={styles.loadingMoreContainer}>
                            <div style={styles.spinner}></div>
                            <p style={styles.loadingText}>Loading more blocks...</p>
                        </div>
                    )}
                </>
            )}
        </div>
    );
};

import { getApiUrl, CONFIG } from '../config/config';
import { RootBlock, RootBlocksQuery } from '../models/block';

export class BlockService {
    static async getRootBlocks(query: RootBlocksQuery = {}): Promise<RootBlock[]> {
        const params = new URLSearchParams();

        if (query.limit !== undefined) {
            params.append('limit', query.limit.toString());
        } else {
            params.append('limit', CONFIG.DEFAULT_PAGE_SIZE.toString());
        }

        if (query.cursor) {
            params.append('cursor', query.cursor);
        }

        const url = getApiUrl(`/root-blocks?${params.toString()}`);

        try {
            const response = await fetch(url);

            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }

            const data: RootBlock[] = await response.json();
            return data;
        } catch (error) {
            console.error('Error fetching root blocks:', error);
            throw new Error('Failed to fetch root blocks');
        }
    }

    static formatDateTime(timestamp: number): string {
        return new Date(timestamp).toLocaleString();
    }

    static formatDuration(startTime: number, endTime: number): string {
        const duration = endTime - startTime;
        if (duration < 1000) {
            return `${duration}ms`;
        }
        return `${(duration / 1000).toFixed(2)}s`;
    }
}
